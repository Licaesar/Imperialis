### Register the Blood (dinasta de dinastia com Sanguinis vivo): o Liber Sanguinis, a cada 25 anos por dinastia
ci_register_blood_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds"
	}
	desc = ci_register_blood_decision_desc
	selection_tooltip = ci_register_blood_decision_tooltip
	confirm_text = ci_register_blood_decision_confirm
	sort_order = 65

	is_shown = {
		has_dynasty = yes
		this = dynasty.dynast
		dynasty = {
			any_dynasty_member = {
				is_alive = yes
				OR = { has_trait = sanguini_1 has_trait = sanguini_2 has_trait = sanguini_3 has_trait = sanguini_4 has_trait = sanguini_5 }
			}
		}
	}

	is_valid = {
		custom_tooltip = {
			text = ci_register_blood_recent_tt
			dynasty = { NOT = { has_variable = ci_blood_registered } }
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
	}

	cost = {
		gold = 200
	}

	effect = {
		dynasty = {
			add_dynasty_prestige = 500
			set_variable = {
				name = ci_blood_registered
				years = 25
			}
		}
		custom_tooltip = ci_register_blood_members_tt
		hidden_effect = {
			dynasty = {
				every_dynasty_member = {
					limit = {
						is_alive = yes
						OR = { has_trait = sanguini_1 has_trait = sanguini_2 has_trait = sanguini_3 has_trait = sanguini_4 has_trait = sanguini_5 }
						NOT = { has_character_modifier = ci_liber_sanguinis_modifier }
					}
					add_character_modifier = ci_liber_sanguinis_modifier
				}
			}
		}
	}

	ai_check_interval = 60

	ai_will_do = {
		base = 50
	}
}
