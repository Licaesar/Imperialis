		}
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_confession.dds"

		barony = 0

		empire = 60
		hegemony = 60
	}

	sort_order = 110
	
	is_shown = {
		has_spiritual_fulfillment_type = ci_imperial_fulfillment # Imperialis
		# Imperialis (4d, decisao do autor): sem o tenet Confession, que nenhum rito do Imperialis tem
		has_pam_dlc_trigger = yes
	}

	is_valid = {
		NOT = { has_trait = excommunicated }
		custom_tooltip = {
			text = ci_pam_decision_confession_sp_court_chaplain

		}
	}

		is_alive = yes
		is_in_activity = no
		is_imprisoned = no
		cp:councillor_court_chaplain ?= {
			is_available = yes
		}
	}

	cooldown = { years = 5 }

	effect = {
		if = {
			limit = { spiritual_fulfillment < 0 }
			custom_tooltip = ci_pam_decision_confession_tooltip_sp_effect
		}
		custom_tooltip = ci_pam_decision_confession_tooltip_stress_effect
		custom_tooltip = ci_pam_decision_events.0001.a.confession_random_outcome
		trigger_event = ci_pam_decision_events.0001
	}

	ai_potential = {
		stress_level >= 1
	}

	ai_will_do = {
		base = 35
		modifier = {
			stress_level >= 2
			add = 30
		}
		modifier = {
			stress_level >= 3
			add = 30
		}
		modifier = {
			has_trait = humble
			add = 20
		}
		modifier = {
			has_trait = zealous
			add = 20
		}
		modifier = {
			has_trait = cynical
			add = -40
		}
	}
}

			text = cardinal_unlocks_pam_promulgate_word_of_pious_deeds_decision_tooltip

			text = domicile_unlocks_pam_promulgate_word_of_pious_deeds_decision_tooltip

		is_available_quick = {

			at_war = no
			imprisoned = no
			activity = no
		}
	}

	cost = {
		piety = massive_piety_value
	}

	effect = {
		add_character_flag = pam_promulgate_word_of_pious_deeds_used

			limit = {
				exists = superior.primary_title
			}

				save_scope_as = praise_worthy_priest_superior_title
			}

			}
		}

				save_scope_as = praise_worthy_priest_superior_title
			}
		}

ci_pam_cardinal_endow_masses_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_papal_petition.dds"
	}
	decision_group_type = religious

	desc = ci_pam_cardinal_endow_masses_decision_desc
	selection_tooltip = ci_pam_cardinal_endow_masses_decision_tooltip
	confirm_text = ci_pam_cardinal_endow_masses_decision_confirm

	ai_check_interval_by_tier = {
		barony = 0
		county = 24
		duchy = 24
		kingdom = 24
		empire = 24
		hegemony = 24
	}

	sort_order = 105

	is_shown = {
		is_clergy = yes
		has_spiritual_fulfillment_type = ci_imperial_fulfillment # Imperialis
		has_pam_dlc_trigger = yes
	}

	is_valid = {
		#Check if they are a Cardinal
		custom_tooltip = {
			text = cardinal_unlocks_pam_cardinal_endow_masses_decision_tooltip
			exists = root.clerical_elector_title
		}
	}

	is_valid_showing_failures_only = {
		is_available_quick = {
			incapable = no
			alive = yes
			adult = yes
			imprisoned = no
			activity = no
		}
	}

	cooldown = { years = 5 }

	cost = {
		treasury_or_gold = medium_treasury_or_gold_value
	}

	effect = {
		send_interface_toast = {
			title = ci_pam_cardinal_endow_masses_decision_toast
			left_icon = root
			add_piety = major_piety_gain
			# Simony holds the exchange of wealth for sacred benefit to be the faith working as intended
			if = {
				limit = {
					OR = {
						rite = { rite_has_tenet = tenet_simony }
						has_personal_tenet = tenet_simony
					}
				}
				change_spiritual_fulfillment = medium_spiritual_fulfillment_value
			}
			custom_tooltip = {
				text = ci_pam_cardinal_endow_masses_decision_subordinates_tt
				every_subordinate = {
					add_piety = minor_piety_gain
					add_opinion = {
						target = root
						modifier = ci_pam_cardinal_endowed_masses_opinion
					}
				}
			}
		}
	}

	ai_potential = {
		exists = root.clerical_elector_title
		# A cardinal holding anything beyond the cardinalate has better uses for the treasury
		NOT = {
			any_held_title = {
				this != root.clerical_elector_title
			}
		}
		trigger_if = {
			limit = { has_treasury = yes }
			short_term_treasury >= medium_treasury_or_gold_value
		}
	}

	ai_will_do = {
		base = 30
		modifier = {
			has_trait = zealous
			add = 25
		}
		modifier = {
			piety_level < 3
			add = 20
		}
	}
}
