# Override de great_project_cathedral_01, _02 e _03 (common/buildings/pam_buildings.txt, CK3 1.20.1), copias do vanilla. Mudanca
# marcada com 'Imperialis' (marco 3, fase 3b.2): so o is_enabled aceita tambem ci_imperial_fulfillment; sem isso a catedral pronta
# ficaria desligada (sem os modificadores de piedade, opiniao e desenvolvimento) para o titular Imperialis. As constantes @illustration_*
# sao locais do arquivo vanilla e vao junto. Gerado por docs/dev/regen_cathedral_overrides.py; regerar a cada atualizacao do jogo.

@illustration_catholic_grand_cathedral = "gfx/interface/illustrations/holding_types/pam_temple_romanesque.dds"
@illustration_catholic_gothic_cathedral = "gfx/interface/illustrations/holding_types/pam_temple_gothic.dds"
@illustration_orthodox_grand_cathedral = "gfx/interface/illustrations/holding_types/temple_orthodox.dds"

great_project_cathedral_01 = {
	effect_desc = {
		desc = cathedral_domicile_desc
	}

	type_icon = "icon_structure_cathedral.dds"
	#Catholic
		asset = {
			type = pdxmesh
			name = "pam_generic_romanesque_01_a_mesh"
			requires_dlc_flag = by_god_alone
			illustration = @illustration_catholic_grand_cathedral
			soundeffect = { soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/Temples/church_temple" soundparameter = { "Tier" = 0 } }
			graphical_faiths = { "catholic_gfx" } #+1
			graphical_regions = { graphical_western graphical_mena graphical_india graphical_mediterranean graphical_steppe graphical_east_asia graphical_siberia } #+4
		}
	#Orthodox
	asset = {
		type = pdxmesh
		name = "pam_generic_orthodox_01_a_mesh"
		requires_dlc_flag = by_god_alone
		illustration = @illustration_orthodox_grand_cathedral
		soundeffect = { soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/Castle/mediterranean_castle" soundparameter = { "Tier" = 0 } }
		graphical_cultures = { byzantine_building_gfx caucasian_building_gfx }
		graphical_faiths = { "orthodox_gfx" }
		graphical_regions = { graphical_western graphical_mena graphical_india graphical_mediterranean graphical_steppe graphical_east_asia graphical_siberia }
	}
	asset = {
		type = pdxmesh
		name = "pam_generic_romanesque_01_a_mesh"
		requires_dlc_flag = by_god_alone
		illustration = @illustration_catholic_grand_cathedral
		soundeffect = { soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/Temples/church_temple" soundparameter = { "Tier" = 0 } }
		graphical_regions = { graphical_western graphical_mena graphical_india graphical_mediterranean graphical_steppe graphical_east_asia graphical_siberia }
	}

	#We only build this through Decision -> Great Projects
	can_construct_potential = {
		always = no #This should never show up
	}

	# Imperialis (marco 3, 3b.2): os modificadores da catedral tambem valem para a realizacao espiritual propria do Imperialis
	is_enabled = {
		scope:holder = {
			OR = {
				has_spiritual_fulfillment_type = christian_fulfillment
				has_spiritual_fulfillment_type = ci_imperial_fulfillment
			}
			government_has_flag = government_is_ecclesiastical
		}
	}

	show_disabled = yes

	character_modifier = {
		monthly_piety = 0.25
		zealot_opinion = 10
	}
	
	county_modifier = {
		development_growth_factor = 0.1
		monthly_county_control_growth_factor = 0.05
		tax_mult = 0.1
	}

	on_complete = {	
		every_province_domicile = {
			limit = {
				is_domicile_type = ecclesiastical_domicile
				# Imperialis (marco 4, 4a): corrige o erro vanilla de construir de novo o mesmo nivel do dominio
				NOT = { has_domicile_building_or_higher = ecclesiastical_domicile_main_02 }
				NOT = { has_domicile_construction = ecclesiastical_domicile_main_02 }
			} 
			# Imperialis (pos-1.10): com outra obra em andamento a versao animada falha; entao sobe o nivel direto
			if = {
				limit = { has_ongoing_domicile_construction = no }
				add_animate_domicile_building = ecclesiastical_domicile_main_02
			}
			else = {
				add_domicile_building = ecclesiastical_domicile_main_02
			}
		}
	}
	
	ai_value = {
		base = 100
		ai_pious_building_preference_modifier = yes
		modifier = { # Fill all building slots before going for special buildings
			factor = 0
			free_building_slots > 0
		}
	}
	
	type = great_building

	flag = grand_cathedral_flag
	flag = great_project_cathedral_building
}
