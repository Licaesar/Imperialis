# Na 1.20 a doutrina aponta para o grupo com doctrine_group_type; o grupo nao lista mais as doutrinas.
# O grupo dos Ordo (doctrine_imperialis_branches) saiu em 11/10/2026, junto com as doutrinas de identidade dos ritos (decisao do autor).

# Regioes clericais da fe (marco 3, fase 3b.1). Como o is_christian_faith do vanilla (categoria special): a doutrina vai junto para a fe
# nova criada por divergencia. A doutrina de identidade crista nao serve (traz sincretismo, hostilidade e a situacao da Igreja crista).
ci_has_clerical_regions = {
	category = special

	is_available_on_create = {
		has_doctrine = ci_doctrine_has_clerical_regions
	}
}

# Bencao do Ministorum (marco 4, fase 4a): como o grupo vanilla pope_blessing (Holy Blessing dos ritos catolicos). Travado no rito e
# nunca escolhido na criacao de fe; os ritos Sororitas, Omnissiah e Homus o carregam desde o inicio.
ci_ministorum_blessing_group = {
	category = special

	doctrine_lock = rite

	is_available_on_create = {
		always = no
	}
}

# Teachings of the Omnissiah (marco 4, fase 4c): como o grupo vanilla is_christian_faith (Teachings of Jesus), categoria special e
# visivel. A doutrina fica na religiao (vale para os quatro ritos) e vai junto para a fe nova criada por divergencia, como no vanilla.
ci_teachings_of_the_omnissiah = {
	category = special

	is_available_on_create = {
		has_doctrine = ci_doctrine_teachings_of_the_omnissiah
	}
}

# Concordia Omnissiae (marco 5, fase 5b): categoria special e visivel. Travado na fe: um rito que diverge ate virar fe nova perde a
# doutrina e fica hostil as duas fes da religiao. Ritos novos dentro da mesma fe a mantem.
ci_omnissiae_concord = {
	category = special

	doctrine_lock = faith

	is_available_on_create = {
		has_doctrine = ci_doctrine_omnissiae_concord
	}
}

# Veneratio Sanguinis (11/10/2026): categoria special e visivel, como a Teachings of the Omnissiah. Vai junto para a fe nova criada por
# divergencia de um rito da Veritas.
ci_veneratio_sanguinis = {
	category = special

	is_available_on_create = {
		has_doctrine = ci_doctrine_veneratio_sanguinis
	}
}
