##############################################
# CULTUS IMPERIALIS - TENETS ci_ (marco 2, fase b)
#
# Versoes imperiais dos tenets cristaos da By God Alone. Cada tenet clona a estrutura do original (parametros, modificadores
# e virtudes/pecados iguais), trocando so a trava de religiao: vale para imperialis_religion em vez de christianity_religion.
# O script do jogo le os PARAMETROS (has_doctrine_parameter / has_personal_tenet_flag), por isso os clones funcionam sem mais codigo.
# O que o jogo liga a CHAVE do tenet (eventos de concilio, de peticao ao Head of Faith, evangelizacao) nao e herdado.
#
# Todos entram como permitidos nos quatro ritos (history/faiths/ci_faith_history.txt): sao escolha de tenet pessoal e somam
# virtudes e pecados com peso 0.5 (o personal tenet vale em dobro). Parametros e modificadores de fe so valem se um rito
# os adotar como central. A linha "clone:" abaixo e lida por docs/dev/check_imperialis_religion.py para conferir que os
# parametros continuam iguais aos do original.
#
# Nomes provisorios (TODO_LORE): ver localization/english/ci_tenets_l_english.yml.
##############################################

# clone: tenet_dulia
tenet_ci_ancestor_veneration = {
	icon = tenet_dulia

	requires_dlc_flag = by_god_alone

	is_shown = {
		religion = religion:imperialis_religion
	}

	can_pick = {
		religion = religion:imperialis_religion
	}

	can_pick_as_personal_tenet = {
		faith = { religion = religion:imperialis_religion }
	}

	piety_cost = {
		value = faith_tenet_cost_low

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_ci_ancestor_veneration }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {
		tenet_dulia_cheaper_saint_costs
		tenet_dulia_extra_relic_modifier
		tenet_dulia_dynasty_prestige_from_relics
		tenet_dulia_canonization_faith_bonus
	}

	personal_tenet_parameters = {
		tenet_dulia_can_select_patron_saint
		tenet_dulia_personal_sainthood_chance
	}

	personal_tenet_modifier = {
		artifact_decay_reduction_mult = -0.25
	}

	traits = {
		virtues = {
			consecrated_blood = { weight = 15 scale = 2 }
			zealous = { weight = 10 scale = 1 }
		}
		sins = {
			cynical = { weight = 10 scale = 1 }
		}
	}
}
