##############################################
# CULTUS IMPERIALIS - TENETS ci_ (marco 2, fase b)
#
# Tres tenets-base que o autor vai reescrever para virarem tenets unicos da fe Imperialis. Hoje cada um ainda clona o tenet
# cristao da By God Alone (mesmos parametros, modificadores e virtudes/pecados); o script do jogo le os PARAMETROS
# (has_doctrine_parameter / has_personal_tenet_flag), por isso funcionam sem codigo extra. O que o jogo liga a CHAVE do tenet
# vanilla (eventos de concilio, de peticao ao Head of Faith, evangelizacao) nao e herdado.
#
# Atencao: o jogo deixa QUALQUER personagem que conheca um tenet adota-lo como pessoal, mesmo de outra religiao (jogaveis
# muculmanos e pagaos usam tenets cristaos da DLC). Por isso cada tenet daqui tem can_pick_as_personal_tenet restrito a
# imperialis_religion. Mantenha essa trava ao reescrever.
#
# Todos entram como permitidos nos quatro ritos (history/faiths/ci_faith_history.txt): escolha de tenet pessoal e virtudes e
# pecados com peso 0.5 (o personal tenet vale em dobro). Parametros e modificadores de fe so valem se um rito os adotar como
# central. O Head of Faith (decisoes vanilla pam_permit_tenet_decision e pam_prohibit_tenet_decision) pode permitir ou proibir
# tenets da fe: no teste de 899, dois dos tres ficaram proibidos nos quatro ritos.
#
# A linha "# clone:" e lida por docs/dev/check_imperialis_religion.py, que AVISA (nao falha) quando um tenet passa a diferir
# do original; apague a linha de um tenet quando ele virar totalmente proprio.
#
# Nomes provisorios (TODO_LORE): ver localization/english/ci_tenets_l_english.yml.
##############################################

# clone: tenet_celestial_hierarchy
tenet_ci_sanguini_hierarchy = {
	icon = tenet_celestial_hierarchy

	requires_dlc_flag = by_god_alone

	is_shown = {
		religion = religion:imperialis_religion
	}

	can_pick = {
		religion = religion:imperialis_religion
	}

	can_pick_as_personal_tenet = {
		faith = { religion = religion:imperialis_religion }
	}

	piety_cost = {
		value = faith_tenet_cost_low

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_ci_sanguini_hierarchy }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {
		tenet_celestial_hierarchy_court_aptitude_bonus
		tenet_celestial_hierarchy_clergy_acceptance
		tenet_celestial_hierarchy_religious_relations_bonus
	}

	character_modifier = {
		piety_level_impact_mult = 0.25
	}

	county_modifier = {
		travel_danger = -10
	}

	personal_tenet_parameters = {
		tenet_celestial_hierarchy_sp_from_court_position
		tenet_celestial_hierarchy_interaction_acceptance
	}

	traits = {
		virtues = {
			loyal = { weight = 20 scale = 1 }
			humble = { weight = 15 scale = 1 }
			content = { weight = 15 scale = 1 }
		}
		sins = {
			disloyal = { weight = 20 scale = 1 }
			ambitious = { weight = 15 scale = 1 }
		}
	}
}
