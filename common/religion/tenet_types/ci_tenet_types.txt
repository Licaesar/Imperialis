##############################################
# CULTUS IMPERIALIS - TENETS ci_ (marco 2, fase b)
#
# Dois tenets da fe Imperialis com base nos tenets cristaos da By God Alone (mesmos parametros, modificadores e virtudes/pecados)
# e, desde o marco 4 (fase 4b, aprovado pelo autor em 10/10/2026), um traco exclusivo cada (marcado 'Imperialis'). O script do jogo le os PARAMETROS
# (has_doctrine_parameter / has_personal_tenet_flag), por isso funcionam sem codigo extra. O que o jogo liga a CHAVE do tenet
# vanilla (eventos de concilio, de peticao ao Head of Faith, evangelizacao) nao e herdado.
#
# Atencao: o jogo deixa QUALQUER personagem que conheca um tenet adota-lo como pessoal, mesmo de outra religiao (jogaveis
# muculmanos e pagaos usam tenets cristaos da DLC). Por isso cada tenet daqui tem can_pick_as_personal_tenet restrito a
# imperialis_religion. Mantenha essa trava ao reescrever.
#
# Todos entram como permitidos nos quatro ritos (history/faiths/ci_faith_history.txt): escolha de tenet pessoal e virtudes e
# pecados com peso 0.5 (o personal tenet vale em dobro). Parametros e modificadores de fe so valem se um rito os adotar como
# central. O Head of Faith (decisoes vanilla pam_permit_tenet_decision e pam_prohibit_tenet_decision) pode permitir ou proibir
# tenets da fe: no teste de 899, dois dos antigos ci_ ficaram proibidos nos quatro ritos.
#
# Nomes e descricoes: localization/english/ci_tenets_l_english.yml.
##############################################

# base: tenet_miles_christi
tenet_ci_militarum = {
	icon = tenet_miles_christi

	requires_dlc_flag = by_god_alone

	is_shown = {
		religion = religion:imperialis_religion
	}

	can_pick = {
		religion = religion:imperialis_religion
	}

	can_pick_as_personal_tenet = {
		faith = { religion = religion:imperialis_religion }
	}

	piety_cost = {
		value = faith_tenet_cost_low

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_ci_militarum }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {
		tenet_miles_christi_white_peace_defender_bonus
		tenet_miles_christi_combat_fulfillment
		tenet_miles_christi_holy_weapon_blessing
		sanctify_artifacts_weapons
	}

	character_modifier = {
		prowess_per_piety_level = 1
		knight_limit = 1
		accolade_glory_gain_mult = 0.2
		monthly_piety_gain_per_knight_mult = 0.01
		holy_order_hire_cost_mult = -0.15
		# Imperialis (marco 4, 4b): a infantaria pesada da Legio; as espadas de fogo dos Sanguinis (4e) sao infantaria pesada
		heavy_infantry_damage_mult = 0.1
		heavy_infantry_toughness_mult = 0.1
	}

	personal_tenet_modifier = {
		martial_per_piety_level = 1
		knight_limit = 1
	}

	personal_tenet_parameters = {
		tenet_miles_christi_holy_warrior_trait
	}

	traits = {
		virtues = {
			brave = { weight = 25 scale = 2 }
			faith_warrior = { weight = 25 scale = 2 }
			crusader_king = { weight = 25 scale = 2 }
		}
		sins = {
			craven = { weight = 25 scale = 2 }
		}
	}
}
