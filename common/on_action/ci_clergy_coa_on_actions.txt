on_game_start_after_lobby = {
	on_actions = { ci_clergy_coa_on_game_start }
}

ci_clergy_coa_on_game_start = {
	effect = {
		faith:imperialis = {
			organization ?= { set_coa = ci_veritas_clergy_coa }
		}
		faith:ci_lex_omnissiah = {
			organization ?= { set_coa = ci_lex_clergy_coa }
		}
	}
}
