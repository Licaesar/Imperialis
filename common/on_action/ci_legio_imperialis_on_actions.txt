##############################################
# CULTUS IMPERIALIS - LEGIO IMPERIALIS
# Cria o acampamento de aventureiro sem terra de Kensoriel Liphe em Jerusalem
# e move Helaena, os Apostolos e as esposas para a corte dele.
# Feito por script porque o historico nao permite "employer" para personagem sem terra.
##############################################

# Antes do lobby: cria o acampamento para ele ja aparecer na selecao de personagem.
on_game_start = {
	on_actions = { ci_on_game_start_legio_imperialis }
}

# Depois do lobby o vanilla (game_start.txt, EP3 LAAMP SETUP) expulsa do acampamento
# todo cortesao que nao seja da dinastia do lider. Reunimos a corte de novo em seguida,
# e mais uma vez no dia seguinte por seguranca (o efeito e idempotente).
on_game_start_after_lobby = {
	on_actions = {
		ci_on_game_start_after_lobby_legio_imperialis
		delay = { days = 1 }
		ci_on_game_start_after_lobby_legio_imperialis
	}
}

ci_on_game_start_legio_imperialis = {
	effect = {
		character:ci_kensoriel_liphe ?= {
			if = {
				limit = {
					is_alive = yes
					is_ruler = no
					has_dlc_feature = roads_to_power
				}
				save_scope_as = new_landless_adventurer

				# Coloca Kensoriel em Jerusalem antes de criar o acampamento
				move_to_pool_at = province:5965 # c_jerusalem

				# Titulo do acampamento
				create_adventurer_title = {
					name = ci_legio_imperialis
					holder = scope:new_landless_adventurer
					article = ci_legio_imperialis_article
					save_scope_as = ci_legio_imperialis_title
				}
				if = {
					limit = { NOT = { has_government = landless_adventurer_government } }
					change_government = landless_adventurer_government
				}
				if = {
					limit = { NOT = { has_realm_law = landless_adventurer_succession_law } }
					add_realm_law = landless_adventurer_succession_law
				}

				# Acampamento em c_jerusalem
				domicile ?= {
					move_domicile = province:5965
					change_provisions = starting_provisions_value
				}
			}
		}
		ci_legio_imperialis_gather_effect = yes
	}
}

ci_on_game_start_after_lobby_legio_imperialis = {
	effect = {
		ci_legio_imperialis_gather_effect = yes
	}
}
