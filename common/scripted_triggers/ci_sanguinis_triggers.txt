# Cura dos Sanguinis (marco 4, fase 4e): vagas do desgaste "Spent Blood" (ate 3 ao mesmo tempo).

# A vaga $SLOT$ (1 a 3) esta ocupada, em qualquer intensidade.
ci_sanguinis_spent_blood_slot_trigger = {
	OR = {
		has_character_modifier = ci_sanguinis_spent_blood_$SLOT$_075
		has_character_modifier = ci_sanguinis_spent_blood_$SLOT$_050
		has_character_modifier = ci_sanguinis_spent_blood_$SLOT$_025
	}
}

# As tres vagas estao ocupadas: a cura fica bloqueada (decisao do autor).
ci_sanguinis_healing_strain_full_trigger = {
	ci_sanguinis_spent_blood_slot_trigger = { SLOT = 1 }
	ci_sanguinis_spent_blood_slot_trigger = { SLOT = 2 }
	ci_sanguinis_spent_blood_slot_trigger = { SLOT = 3 }
}
