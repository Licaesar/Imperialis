# Triggers da decisao que funda o Adeptus Ministorum (k_sol).

# Escopo: personagem. Helaena Undomiel e a primeira titular quando esta viva, adulta e seguidora da fe imperialis.
ci_ministorum_helaena_available_trigger = {
	exists = character:ci_helaena_undomiel
	character:ci_helaena_undomiel = {
		is_alive = yes
		is_adult = yes
		faith = faith:imperialis
	}
}

# Escopo: personagem. Candidato a titular na falta de Helaena: seguidor adulto da fe (qualquer rito).
ci_ministorum_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	faith = faith:imperialis
}

# Escopo: governante que toma a decisao. Ha alguem para ocupar o titulo (Helaena, ou candidato na corte ou entre os vassalos).
ci_ministorum_holder_exists_trigger = {
	OR = {
		ci_ministorum_helaena_available_trigger = yes
		any_courtier_or_guest = { ci_ministorum_candidate_trigger = yes }
		any_vassal = { ci_ministorum_candidate_trigger = yes }
	}
}

# Falta integrar o Ministorum ao imperio? (escopo: o imperador da Italia, quem toma ci_ministorum_vassalage_decision)
# Verdadeiro enquanto o titular do k_sol nao for vassalo do imperador, o d_romagna nao for de jure do k_sol, o k_sol nao for de jure do
# e_italy ou o titular nao tiver o d_et_ravenna. Quando nada falta a decisao de vassalagem some (is_shown) e a IA nao a considera
# (ai_potential). Mesmo conjunto de partes que ci_ministorum_vassalize_effect aplica.
ci_ministorum_integration_pending_trigger = {
	OR = {
		title:k_sol.holder ?= {
			NOT = { liege ?= root }
		}
		NOT = { title:d_romagna.de_jure_liege ?= title:k_sol }
		NOT = { title:k_sol.de_jure_liege ?= title:e_italy }
		NOT = { title:d_et_ravenna.holder ?= title:k_sol.holder }
	}
}
