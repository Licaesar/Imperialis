# Override de is_local_clerical_region_succession_notification_trigger (common/scripted_triggers/00_important_actions_triggers.txt, CK3 1.20.0.3), copia do vanilla.
# Mudanca marcada com 'Imperialis' (marco 4, fase 4a): corrige o escopo do is_head_of_faith dentro do bloco do personagem.
# Gerado por docs/dev/regen_vanilla_fix_overrides.py; regerar a cada atualizacao do jogo.

is_local_clerical_region_succession_notification_trigger = {
	has_clerical_region = yes
	is_head_of_faith = no
	$CHARACTER$.capital_county.clerical_region_title ?= this
	exists = holder
	holder != $CHARACTER$
	$CHARACTER$ = {
		has_pam_dlc_trigger = yes
		is_landed = yes
		government_has_flag = government_is_theocracy
		# Imperialis (marco 4, 4a): o vanilla usa aqui is_head_of_faith, que so vale para titulo; o teste de personagem equivalente:
		NOT = { faith.religious_head ?= this }
		NOT = {
			any_held_title = {
				has_clerical_region = yes
			}
		}
	}
	place_in_line_of_succession = {
		target = $CHARACTER$
		value <= 3
	}
}
