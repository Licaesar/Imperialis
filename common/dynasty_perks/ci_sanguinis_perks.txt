ci_sanguinis_legacy_1 = { # Vigor Sanguinis
	legacy = ci_sanguinis_legacy_track

	can_be_picked = {
		dynasty ?= {
			any_dynasty_member = {
				is_alive = yes
				OR = { has_trait = sanguini_1 has_trait = sanguini_2 has_trait = sanguini_3 has_trait = sanguini_4 has_trait = sanguini_5 }
			}
		}
	}

	character_modifier = {
		health = 0.25
		fertility = 0.15
	}

	ai_chance = {
		value = 100
		if = {
			limit = { can_start_new_legacy_track_trigger = no }
			multiply = 0
		}
	}
}
