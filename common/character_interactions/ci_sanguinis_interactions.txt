# Cura dos Sanguinis (marco 4, fase 4e; aprovado pelo autor em 10/10/2026). Um Sanguinis nivel 4 ou 5 cura todas as doencas (traços
# com a marca illness) de outro personagem, de qualquer fe. Sem recarga e sem limite (pedido do autor no reteste de 10/10/2026): cada
# cura soma um "Spent Blood" (desgaste de saude por 5 anos, acumulavel). A IA nao usa (decisao do autor).
# Efeitos em common/scripted_effects/ci_sanguinis_effects.txt.
ci_sanguinis_healing_interaction = {
	category = interaction_category_friendly
	icon = plague
	interface_priority = 30

	use_diplomatic_range = yes
	desc = ci_sanguinis_healing_interaction_desc
	greeting = positive
	auto_accept = yes

	is_shown = {
		scope:actor != scope:recipient
		scope:actor = {
			OR = {
				has_trait = sanguini_4
				has_trait = sanguini_5
			}
		}
	}

	# Playtest de 10/10/2026: a cura aparece para qualquer personagem (categoria Personal); sem doenca, fica bloqueada com aviso.
	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = ci_sanguinis_healing_no_illness_tt
				has_trait_with_flag = illness
			}
		}
		scope:actor = { is_available_adult = yes }
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = ci_sanguinis_healing_interaction_actor_toast
				left_icon = scope:actor
				right_icon = scope:recipient
				scope:recipient = {
					ci_sanguinis_heal_effect = yes
					add_opinion = {
						target = scope:actor
						modifier = ci_sanguinis_healed_opinion
					}
				}
				ci_sanguinis_add_healing_strain_effect = yes
			}
			# Primeira cura deste Sanguinis (marco 5, fase 5d): evento de lore uma vez por personagem (a IA nao cura)
			if = {
				limit = { NOT = { has_character_flag = ci_sanguinis_first_cure_done } }
				add_character_flag = ci_sanguinis_first_cure_done
				scope:recipient = { save_scope_as = ci_lore_healed }
				trigger_event = ci_lore.0050
			}
		}
		scope:recipient = {
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_good
					title = ci_sanguinis_healing_interaction_recipient_toast
					desc = ci_sanguinis_healing_interaction_recipient_toast_desc
					left_icon = scope:recipient
					right_icon = scope:actor
				}
			}
		}
	}

	# A IA nao usa a cura (decisao do autor, 10/10/2026): sem ai_frequency nem ai_targets, a IA nunca considera a interacao
}
