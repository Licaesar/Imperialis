# Amizades dos fundadores (chamado no historico de Kensoriel):
# cada Apostolo e amigo de Kensoriel, de Helaena e de todos os outros Apostolos.
ci_founders_friendships_effect = {
	character:ci_reid_astrea ?= { add_to_list = ci_apostoli }
	character:ci_ezio_auditore ?= { add_to_list = ci_apostoli }
	character:ci_emhyr_emreis ?= { add_to_list = ci_apostoli }
	character:ci_vistarios_granhieron ?= { add_to_list = ci_apostoli }
	character:ci_legarde_martyr ?= { add_to_list = ci_apostoli }
	character:ci_hrodric_gyliman ?= { add_to_list = ci_apostoli }
	character:ci_verso_lumere ?= { add_to_list = ci_apostoli }
	character:ci_altair_ramashiri ?= { add_to_list = ci_apostoli }
	character:ci_sigvardr_drakenfyr ?= { add_to_list = ci_apostoli }
	character:ci_miyamoto_musashi ?= { add_to_list = ci_apostoli }
	character:ci_aurelian_solerius ?= { add_to_list = ci_apostoli }
	character:ci_lakan_shaokan ?= { add_to_list = ci_apostoli }
	character:ci_astarinus_viriata ?= { add_to_list = ci_apostoli }
	character:ci_julius_euclius ?= { add_to_list = ci_apostoli }
	character:ci_radahn_varakh ?= { add_to_list = ci_apostoli }
	character:ci_alia_atreides ?= { add_to_list = ci_apostoli }
	character:ci_sun_tzu ?= { add_to_list = ci_apostoli }

	every_in_list = {
		list = ci_apostoli
		save_scope_as = ci_apostle
		character:ci_kensoriel_liphe ?= {
			if = {
				limit = { NOT = { has_relation_friend = scope:ci_apostle } }
				set_relation_friend = { reason = friend_generic_history target = scope:ci_apostle }
			}
		}
		character:ci_helaena_undomiel ?= {
			if = {
				limit = { NOT = { has_relation_friend = scope:ci_apostle } }
				set_relation_friend = { reason = friend_generic_history target = scope:ci_apostle }
			}
		}
		every_in_list = {
			list = ci_apostoli
			limit = {
				NOT = { this = scope:ci_apostle }
				NOT = { has_relation_friend = scope:ci_apostle }
			}
			set_relation_friend = { reason = friend_generic_history target = scope:ci_apostle }
		}
	}
}
