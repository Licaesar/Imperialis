# Lendarios (qualidade e riqueza 100, como set_artifact_rarity_illustrious do vanilla), sem desgaste, modelos 3D do vanilla.

# Escopo: o dono. KEY - chave curta (nome ci_relic_<KEY>_name, descricao ci_relic_<KEY>_desc, modificador ci_relic_<KEY>_modifier)
ci_create_founder_relic_effect = {
	save_scope_as = ci_relic_owner
	create_artifact = {
		name = ci_relic_$KEY$_name
		description = ci_relic_$KEY$_desc
		type = $TYPE$
		visuals = $VISUALS$
		modifier = ci_relic_$KEY$_modifier
		template = ci_founder_relic_template
		wealth = 100
		quality = 100
		decaying = no
		history = {
			type = created_before_history
		}
	}
}
