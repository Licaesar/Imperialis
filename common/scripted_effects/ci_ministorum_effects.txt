# Funda o Adeptus Ministorum: cria o titulo k_sol e o entrega a primeira titular.
# Escopo: governante que toma a decisao (ci_found_ministorum_decision).
# Primeira titular: Helaena Undomiel, se viva, adulta e da fe imperialis. Senao, o seguidor adulto da fe com a maior soma de
# aprendizado e nivel de piedade entre a corte (cortesaos e convidados) e os vassalos diretos do decisor.
ci_found_ministorum_effect = {
	save_scope_as = ci_ministorum_founder

	if = {
		limit = { ci_ministorum_helaena_available_trigger = yes }
		character:ci_helaena_undomiel = { save_scope_as = ci_ministorum_first_head }
	}
	else = {
		every_courtier_or_guest = {
			limit = { ci_ministorum_candidate_trigger = yes }
			add_to_temporary_list = ci_ministorum_candidates
		}
		every_vassal = {
			limit = { ci_ministorum_candidate_trigger = yes }
			add_to_temporary_list = ci_ministorum_candidates
		}
		ordered_in_list = {
			list = ci_ministorum_candidates
			order_by = ci_ministorum_candidate_score
			max = 1
			save_scope_as = ci_ministorum_first_head
		}
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = ci_ministorum_change
		add_claim_on_loss = no
	}
	title:k_sol = {
		change_title_holder = {
			holder = scope:ci_ministorum_first_head
			change = scope:ci_ministorum_change
		}
	}
	resolve_title_and_vassal_change = scope:ci_ministorum_change
}
