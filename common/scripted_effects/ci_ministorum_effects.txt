# Funda o Adeptus Ministorum: cria o titulo k_sol, o entrega a primeira titular e o torna o Head of Faith da fe imperialis.
# Escopo: governante que toma a decisao (ci_found_ministorum_decision).
# Primeira titular: Helaena Undomiel, se viva, adulta e da fe imperialis. Senao, o seguidor adulto da fe com a maior soma de
# aprendizado e nivel de piedade entre a corte (cortesaos e convidados) e os vassalos diretos do decisor.
ci_found_ministorum_effect = {
	save_scope_as = ci_ministorum_founder

	if = {
		limit = { ci_ministorum_helaena_available_trigger = yes }
		character:ci_helaena_undomiel = { save_scope_as = ci_ministorum_first_head }
	}
	else = {
		every_courtier_or_guest = {
			limit = { ci_ministorum_candidate_trigger = yes }
			add_to_temporary_list = ci_ministorum_candidates
		}
		every_vassal = {
			limit = { ci_ministorum_candidate_trigger = yes }
			add_to_temporary_list = ci_ministorum_candidates
		}
		ordered_in_list = {
			list = ci_ministorum_candidates
			order_by = ci_ministorum_candidate_score
			max = 1
			save_scope_as = ci_ministorum_first_head
		}
	}

	# ORDEM: condados, ducado, reino (playtest de 04/10/2026: o k_sol e o d_romagna eram criados e destruidos no mesmo dia, e a decisao
	# reaparecia). Um titulo de ducado ou mais que nao e landless nao pode ficar com quem nao tem terras: o jogo o destroi na criacao.
	# So depois de o titular ja ter os condados e que o ducado e o reino nascem; nas maos de quem ja tinha terras, na segunda vez,
	# sobreviveram.
	# Terras do Ministorum (marco 3, 3a; decisao do autor, 04/10/2026): o titular recebe o ducado de Romagna e todos os condados dele,
	# de QUEM OS TIVER (vassalo do fundador, de outro reino ou independente): ninguem fica de fora, e os condes perdem o titulo. As
	# baronias que tinham vassalos ficam com eles, agora vassalos do titular (sem take_baronies: o governo eclesiastico nao pode ter
	# cidades como terra direta, e o estado montado por debug no playtest, sem as cidades, funcionou). O titular fica independente:
	# tornar o k_sol vassalo do imperador, e dar-lhe a Romagna de jure, e a decisao ci_ministorum_vassalage_decision.
	# 1. Os condados.
	create_title_and_vassal_change = {
		type = granted
		save_scope_as = ci_ministorum_counties_change
		add_claim_on_loss = no
	}
	title:d_romagna = {
		every_de_jure_county = {
			limit = {
				exists = holder
				NOT = { holder = scope:ci_ministorum_first_head }
			}
			change_title_holder = {
				holder = scope:ci_ministorum_first_head
				change = scope:ci_ministorum_counties_change
			}
		}
	}
	resolve_title_and_vassal_change = scope:ci_ministorum_counties_change
	# 2. O ducado. Ducados so existem depois de criados: se d_romagna ainda nao foi, nasce ja nas maos do titular.
	if = {
		limit = { title:d_romagna = { is_title_created = no } }
		create_title_and_vassal_change = {
			type = created
			save_scope_as = ci_ministorum_duchy_created
			add_claim_on_loss = no
		}
		title:d_romagna = {
			change_title_holder = {
				holder = scope:ci_ministorum_first_head
				change = scope:ci_ministorum_duchy_created
			}
		}
		resolve_title_and_vassal_change = scope:ci_ministorum_duchy_created
	}
	else_if = {
		limit = {
			title:d_romagna = {
				exists = holder
				NOT = { holder = scope:ci_ministorum_first_head }
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = ci_ministorum_duchy_change
			add_claim_on_loss = no
		}
		title:d_romagna = {
			change_title_holder = {
				holder = scope:ci_ministorum_first_head
				change = scope:ci_ministorum_duchy_change
			}
		}
		resolve_title_and_vassal_change = scope:ci_ministorum_duchy_change
	}
	# 3. O reino.
	create_title_and_vassal_change = {
		type = created
		save_scope_as = ci_ministorum_change
		add_claim_on_loss = no
	}
	title:k_sol = {
		change_title_holder = {
			holder = scope:ci_ministorum_first_head
			change = scope:ci_ministorum_change
		}
	}
	resolve_title_and_vassal_change = scope:ci_ministorum_change

	# O Ministorum passa a ser a chefia da fe (Head of Faith), como o restore_papacy_scripted_effect vanilla faz com k_papal_state.
	# Isso precisa vir antes das leis de sucessao: theocratic_elective_succession_law so pode ser mantida por quem e religious_head.
	faith:imperialis = {
		set_religious_head_title = title:k_sol
	}

	scope:ci_ministorum_first_head = {
		hidden_effect = {
			# Governo eclesiastico (marco 3, 3a): o mesmo do clero com terras e dos cardeais da fe (theocracy_government_type da religiao).
			# Vem antes das leis: trocar de governo pode refazer as leis de sucessao.
			if = {
				limit = { NOT = { government_has_flag = government_is_ecclesiastical } }
				change_government = ecclesiastical_government
			}
			# Sucessao por eleicao do colegio de cardeais (theocratic_elective). O jogo aplicava sozinho tambem a lei teocratica padrao
			# (same_faith_theocratic_succession_law, grupo succession_rel_head) e, com as duas, o titular ficava com
			# succession_result_type = generate_from_template em vez de election (visto no save de 892, playtest de 04/10/2026):
			# common/laws/zzz_ci_law_overrides.txt impede essa lei para fe com colegio.
			if = {
				limit = { NOT = { has_realm_law = theocratic_elective_succession_law } }
				add_realm_law_skip_effects = theocratic_elective_succession_law
			}
		}
	}

	# Preenche o colegio (ate 10; o vanilla so chama isto na criacao de uma fe nova e no inicio do jogo, nao ao criar um Head of Faith).
	# Sem arcebispos para promover, gera os eleitores em Ravena; os cardinalatos dependem dos overrides
	# zzz_ci_cardinal_overrides.txt (scripted_triggers e scripted_effects).
	faith:imperialis = {
		pam_seed_cardinals_effect = yes
	}
}

# Torna o titular do k_sol vassalo de quem toma a decisao ci_ministorum_vassalage_decision (o imperador da Italia) e poe a Romagna de
# jure do k_sol.
# Escopo: o imperador. Mesmo padrao da vassalizacao do vanilla (swear_fealty e change_liege no mesmo lote).
ci_ministorum_vassalize_effect = {
	save_scope_as = ci_ministorum_liege
	title:k_sol.holder = { save_scope_as = ci_ministorum_vassal }
	# Romagna de jure do k_sol (decisao do autor, 04/10/2026): sem isso o rei do k_romagna e o imperador da Italia ficam com base de
	# jure sobre o ducado e os condados, e a IA tenta retira-los do Head of Faith. O k_sol nao tem suserano de jure (de_jure_drift_disabled).
	# Fica aqui, e nao na fundacao, a pedido do autor: o k_sol precisa ja existir e ter dono.
	title:d_romagna = {
		if = {
			limit = { NOT = { de_jure_liege ?= title:k_sol } }
			set_de_jure_liege_title = title:k_sol
		}
	}
	create_title_and_vassal_change = {
		type = swear_fealty
		save_scope_as = ci_ministorum_vassal_change
		add_claim_on_loss = no
	}
	scope:ci_ministorum_vassal = {
		change_liege = {
			liege = scope:ci_ministorum_liege
			change = scope:ci_ministorum_vassal_change
		}
		resolve_title_and_vassal_change = scope:ci_ministorum_vassal_change
	}
}
